"use strict"; // Global State Trackers let searchQuery = ""; let currentGenre = "all"; let cachedAnimeList = []; let selectedAnime = null; let activeEpisodeIdx = 0; let debounceTimer; // Persistent Browser Storage Watchlist Array let localWatchlist = JSON.parse(localStorage.getItem('anime_watchlist')) || []; // UI Selectors const catalogGrid = document.getElementById('catalogGrid'); const homeView = document.getElementById('homeView'); const watchView = document.getElementById('watchView'); const animeSearch = document.getElementById('animeSearch'); const logoHome = document.getElementById('logoHome'); const btnBack = document.getElementById('btnBack'); const streamTitle = document.getElementById('streamTitle'); const mainVideoPlayer = document.getElementById('mainVideoPlayer'); const episodeGrid = document.getElementById('episodeGrid'); const viewHeading = document.getElementById('viewHeading'); const toggleWatchlistBtn = document.getElementById('toggleWatchlistBtn'); const prevEpisodeBtn = document.getElementById('prevEpisodeBtn'); const nextEpisodeBtn = document.getElementById('nextEpisodeBtn'); const screenshotBtn = document.getElementById('screenshotBtn'); // Fetch Global Trends from Jikan API v4 async function fetchTrendingAnime() { catalogGrid.innerHTML = '
Fetching global trends...
'; try { const response = await fetch('https://jikan.moe'); const json = await response.json(); cachedAnimeList = json.data || []; applyFiltersAndRender(); } catch (error) { catalogGrid.innerHTML = '
Failed to load trending items. Rate limit hit.
'; } } // Live Asynchronous Query Search async function searchAnime(query) { if (query.trim().length < 3) return; catalogGrid.innerHTML = '
Searching database...
'; try { const response = await fetch(`https://jikan.moe{encodeURIComponent(query)}&limit=24`); const json = await response.json(); cachedAnimeList = json.data || []; applyFiltersAndRender(); } catch (error) { catalogGrid.innerHTML = '
Error fetching search data.
'; } } // Render Cards with Filter and Watchlist Logic function applyFiltersAndRender() { catalogGrid.innerHTML = ""; const filtered = cachedAnimeList.filter(anime => { if (currentGenre === "watchlist") { return localWatchlist.includes(String(anime.mal_id)); } if (currentGenre === "all") return true; if (!anime.genres) return false; return anime.genres.some(g => g.name.toLowerCase() === currentGenre.toLowerCase()); }); if (filtered.length === 0) { catalogGrid.innerHTML = '
No content fits criteria.
'; return; } filtered.forEach(anime => { const animeId = String(anime.mal_id); const title = anime.title; const image = anime.images?.jpg?.large_image_url || anime.images?.jpg?.image_url || ''; const episodesCount = anime.episodes || '??'; const rating = anime.score ? `★ ${anime.score}` : 'Unrated'; const isSaved = localWatchlist.includes(animeId); const card = document.createElement('div'); card.className = 'anime-card'; card.innerHTML = ` ${isSaved ? '
❤ Saved
' : ''}

${title}

${rating} · ${episodesCount} Eps

`; card.addEventListener('click', () => openPlayer(anime)); catalogGrid.appendChild(card); }); } function openPlayer(anime) { selectedAnime = anime; activeEpisodeIdx = 0; homeView.classList.add('hidden'); watchView.classList.remove('hidden'); streamTitle.textContent = anime.title; updateWatchlistButtonUI(); buildEpisodeButtons(); playEpisode(); } function closePlayer() { mainVideoPlayer.pause(); mainVideoPlayer.src = ""; selectedAnime = null; watchView.classList.add('hidden'); homeView.classList.remove('hidden'); applyFiltersAndRender(); } function updateWatchlistButtonUI() { if (!selectedAnime) return; const animeId = String(selectedAnime.mal_id); const isSaved = localWatchlist.includes(animeId); if (isSaved) { toggleWatchlistBtn.classList.add('in-list'); toggleWatchlistBtn.textContent = '❤ Saved in Watchlist'; } else { toggleWatchlistBtn.classList.remove('in-list'); toggleWatchlistBtn.textContent = '❤ Add to Watchlist'; } } function buildEpisodeButtons() { episodeGrid.innerHTML = ""; const totalEps = selectedAnime.episodes || 12; const renderLimit = Math.min(totalEps, 24); for (let i = 1; i <= renderLimit; i++) { const btn = document.createElement('button'); btn.className = `episode-btn ${i === 1 ? 'active' : ''}`; btn.textContent = i; btn.addEventListener('click', (e) => { activeEpisodeIdx = i - 1; syncEpisodeNavigationUI(); playEpisode(); }); episodeGrid.appendChild(btn); } syncEpisodeNavigationUI(); } function syncEpisodeNavigationUI() { const buttons = document.querySelectorAll('.episode-btn'); buttons.forEach((btn, idx) => { if (idx === activeEpisodeIdx) { btn.classList.add('active'); } else { btn.classList.remove('active'); } }); prevEpisodeBtn.disabled = activeEpisodeIdx === 0; nextEpisodeBtn.disabled = activeEpisodeIdx === (buttons.length - 1); } function playEpisode() { mainVideoPlayer.src = "https://w3schools.com"; mainVideoPlayer.play().catch(() => {}); } // Action Hook Listeners logoHome.addEventListener('click', closePlayer); btnBack.addEventListener('click', closePlayer); animeSearch.addEventListener('input', (e) => { searchQuery = e.target.value; clearTimeout(debounceTimer); if (searchQuery.trim() === "") { viewHeading.textContent = "# Trending Anime"; fetchTrendingAnime(); } else { viewHeading.textContent = "Search Results"; debounceTimer = setTimeout(() => { searchAnime(searchQuery); }, 500); } }); // Watchlist Modification Events toggleWatchlistBtn.addEventListener('click', () => { if (!selectedAnime) return; const animeId = String(selectedAnime.mal_id); const idx = localWatchlist.indexOf(animeId); if (idx > -1) { localWatchlist.splice(idx, 1); } else { localWatchlist.push(animeId); } localStorage.setItem('anime_watchlist', JSON.stringify(localWatchlist)); updateWatchlistButtonUI(); }); // Toolbar Button Streams Control prevEpisodeBtn.addEventListener('click', () => { if (activeEpisodeIdx > 0) { activeEpisodeIdx--; syncEpisodeNavigationUI(); playEpisode(); } }); nextEpisodeBtn.addEventListener('click', () => { const buttons = document.querySelectorAll('.episode-btn'); if (activeEpisodeIdx < buttons.length - 1) { activeEpisodeIdx++; syncEpisodeNavigationUI(); playEpisode(); } }); // Instant Canvas Frame Snapshot Tool screenshotBtn.addEventListener('click', () => { if (mainVideoPlayer.paused || mainVideoPlayer.ended) { alert("Please start playing the video first!"); return; } try { const canvas = document.createElement('canvas'); canvas.width = mainVideoPlayer.videoWidth; canvas.height = mainVideoPlayer.videoHeight; const ctx = canvas.getContext('2d'); ctx.drawImage(mainVideoPlayer, 0, 0, canvas.width, canvas.height); const link = document.createElement('a'); link.download = `${selectedAnime.title}-Snapshot.png`; link.href = canvas.toDataURL(); link.click(); } catch (e) { alert("Browser canvas origin rules restricted capture processing."); } }); // Category Filter Tab Click Events document.querySelectorAll('.filter-btn').forEach(btn => { btn.addEventListener('click', (e) => { document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active')); e.target.classList.add('active'); currentGenre = e.target.getAttribute('data-genre'); applyFiltersAndRender(); }); }); // Launch Application fetchTrendingAnime();
